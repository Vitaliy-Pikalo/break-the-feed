cmake_minimum_required(VERSION 3.16)
project(break_the_feed LANGUAGES CXX)
add_executable(feed src/main.cpp)
target_compile_features(feed PRIVATE cxx_std_17)
enable_testing()
add_test(NAME replay COMMAND feed)
add_executable(feed_check src/check.cpp)
target_compile_features(feed_check PRIVATE cxx_std_17)
add_test(NAME recovery_invariants COMMAND feed_check)
add_executable(feed_fault_check tests/fault_check.cpp)
target_compile_features(feed_fault_check PRIVATE cxx_std_17)
target_include_directories(feed_fault_check PRIVATE src)
add_test(NAME seeded_fault_recovery COMMAND feed_fault_check)
